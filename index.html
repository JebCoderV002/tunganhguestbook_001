Bảng Điều Khiển Admin

Đường dẫn Apps Script sau khi bạn triển khai Google Sheet ở chế độ công khai (Anyone).

Code Google Apps Script Mẫu:

Hãy tạo một Google Sheet mới, vào Tiện ích mở rộng (Extensions) -> Apps Script, dán toàn bộ đoạn code dưới đây vào, đổi tên hàm thành bất kỳ, rồi click Triển khai (Deploy) -> Triển khai mới (New deployment) ở dạng Ứng dụng web (Web app), thiết lập quyền truy cập là Bất kỳ ai (Anyone), sau đó copy đường dẫn web app dán vào ô bên trên.

/* Dán đoạn Code này vào Google Apps Script của bạn */
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Xử lý lưu bút gửi đến
    if (data.action === "submit") {
      var sheet = ss.getSheetByName("Guestbook") || ss.insertSheet("Guestbook");
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(["Thời gian", "Tên người gửi", "Nội dung lưu bút"]);
      }
      sheet.appendRow([new Date(), data.name, data.message]);
      return ContentService.createTextOutput(JSON.stringify({status: "success"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // Xử lý lưu cấu hình (Bảo mật)
    if (data.action === "saveConfig") {
      var sheet = ss.getSheetByName("Config") || ss.insertSheet("Config");
      sheet.clear();
      sheet.appendRow(["Key", "Value"]);
      for (var key in data.config) {
        sheet.appendRow([key, data.config[key]]);
      }
      return ContentService.createTextOutput(JSON.stringify({status: "success"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Config");
    var config = {};
    if (sheet) {
      var data = sheet.getDataRange().getValues();
      for (var i = 1; i < data.length; i++) {
        config[data[i][0]] = data[i][1];
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status: "success", config: config}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
